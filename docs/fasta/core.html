260    def subsample(self, down_to=1, new_path=None, verbose=True):
261        """Pick a given number of sequences from the file pseudo-randomly."""
262        # Pick the destination path #
263        if new_path is None:
264            subsampled = self.__class__(new_temp_path())
265        elif isinstance(new_path, FASTA):
266            subsampled = new_path
267        else:
268            subsampled = self.__class__(new_path)
269        # Check size #
270        if down_to > len(self):
271            message = "Can't subsample %s down to %i. Only down to %i."
272            print(Color.ylw + message % (self, down_to, len(self)) + Color.end)
273            self.copy(new_path)
274            return
275        # Select verbosity #
276        import tqdm
277        if verbose: wrapper = lambda x: tqdm.tqdm(x, total=self.count)
278        else: wrapper = lambda x: x
279        # Generator #
280        def iterator():
281            for read in wrapper(isubsample(self, down_to)):
282                yield read
283        # Do it #
284        subsampled.write(iterator())
285        # Did it work #
286        assert len(subsampled) == down_to
287        # Return #
288        return subsampled

374    def extract_sequences(self, ids,
375                          new_path = None,
376                          in_place = False,
377                          verbose  = False):
378        """
379        Will take all the sequences from the current file who's id appears in
380        the ids given and place them in a new file.
381        If no path is given, a new temporary path is created and returned.
382        If `in_place` is set to True, the original file is removed and replaced
383        with the result of the extraction.
384        Optionally, the argument `ids` can be a function which has to take
385        one string as only input and return True for keeping the sequence and
386        False for discarding the sequence.
387        """
388        # Temporary path #
389        if new_path is None: new_fasta = self.__class__(new_temp_path())
390        elif isinstance(new_path, FASTA): new_fasta = new_path
391        else:                new_fasta = self.__class__(new_path)
392        # Select verbosity #
393        import tqdm
394        wrapper = tqdm.tqdm if verbose else lambda x: x
395        # Simple generator #
396        def simple_match(reads):
397            for r in wrapper(reads):
398                if r.id in ids: yield r
399        # Generator with function #
400        def function_match(reads):
401            for r in wrapper(reads):
402                if ids(r.id): yield r
403        # Do it #
404        if callable(ids):
405            new_fasta.write(function_match(self))
406        else:
407            new_fasta.write(simple_match(self))
408        # Return #
409        if in_place:
410            os.remove(self.path)
411            shutil.move(new_fasta, self.path)
412            return self
413        else: return new_fasta

415    def remove_trailing_stars(self, new_path=None, in_place=True, check=False):
416        """
417        Remove the bad character that can be inserted by some programs at the
418        end of sequences.
419        """
420        # Optional check #
421        if check and int(sh.grep('-c', '\\*', self.path, _ok_code=[0,1])) == 0:
422            return self
423        # Faster with bash utilities #
424        if in_place is True:
425            sh.sed('-i', 's/\\*$//g', self.path)
426            return self
427        # Standard way #
428        if new_path is None: new_fasta = self.__class__(new_temp_path())
429        else:                new_fasta = self.__class__(new_path)
430        new_fasta.create()
431        for seq in self: new_fasta.add_str(str(seq.seq).rstrip('*'), seq.id)
432        new_fasta.close()
433        # Return #
434        return new_fasta

456    def remove_duplicates(self, new_path=None, in_place=True):
457        """
458        If several entries have the same ID in the FASTA file, keep only the
459        first appearance and remove all the others.
460        """
461        # Generator #
462        def unique_entries():
463            seen = set()
464            for i, read in enumerate(self):
465                if read.id in seen: continue
466                else:
467                    seen.add(read.id)
468                    yield read
469        # Return #
470        return self._generator_mod(unique_entries, new_path, in_place)

376                          new_path = None,
377                          in_place = False,
378                          verbose  = False):
379        """
380        Will take all the sequences from the current file who's id appears in
381        the ids given and place them in a new file.
382        If no path is given, a new temporary path is created and returned.
383        If `in_place` is set to True, the original file is removed and replaced
384        with the result of the extraction.

386        one string as only input and return True for keeping the sequence and
387        False for discarding the sequence.

394        import tqdm
395        wrapper = tqdm.tqdm if verbose else lambda x: x
396        # Simple generator #
397        def simple_match(reads):
398            for r in wrapper(reads):
399                if r.id in ids: yield r
400        # Generator with function #
401        def function_match(reads):
402            for r in wrapper(reads):
403                if ids(r.id): yield r
404        # Do it #
