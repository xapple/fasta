 39    def flush(self):
 40        from Bio import AlignIO

 44    def write(self, reads):
 45        from Bio import AlignIO
 46        if not self.directory.exists: self.directory.create()

 51    @cached_property

 55    def gblocks(self,
 56                new_path = None,
 57                seq_type = 'nucl' or 'prot'):
 58        """
 59        Apply the gblocks filtering algorithm to the alignment.
 60        See this page:
 61        http://molevol.cmima.csic.es/castresana/Gblocks/Gblocks_documentation.html
 62        Need to rename all sequences, because it will complain with long names.
 63        """
 64        # Temporary path #
 65        if new_path is None: final = self.__class__(new_temp_path())
 66        else:                final = self.__class__(new_path)
 67        # Mapping every sequence name with a random name #
 68        orig_name_to_temp = {seq.description: 'name' + str(i)
 69                             for i,seq in enumerate(self)}
 70        temp_name_to_orig = {v: k for k, v in orig_name_to_temp.items()}
 71        # Rename every sequence with a random name #
 72        temp_fasta = self.rename_sequences(orig_name_to_temp)
 73        # Options #
 74        if seq_type == 'nucl': t_option = "-t=d"
 75        if seq_type == 'prot': t_option = "-t=p"
 76        # Run it #
 77        result = sh.gblocks91(temp_fasta.path, t_option, '-p=n', "-b4=3",
 78                              "-b3=20", "-b5=a", _ok_code=[0,1])
 79        created_file = temp_fasta.path + '-gb'
 80        assert os.path.exists(created_file)
 81        # Check errors #
 82        if "Execution terminated" in result.stdout:
 83            raise Exception("gblocks crashed again.")
 84        # Back #
 85        temp_fasta.rename_sequences(temp_name_to_orig, final)
 86        # Return #
 87        return final

 89    def build_tree(self, *args, **kwargs):
 90        """
 91        Dispatch a tree build call. Note that you need at least four
 92        taxa to express some evolutionary history on an unrooted tree.
 93        """
 94        # Check length #
 95        assert len(self) > 3
 96        # Default option #
 97        algorithm = kwargs.pop(kwargs, None)
 98        if algorithm is None: algorithm = 'raxml'
 99        # Dispatch #
100        if algorithm == 'raxml':
101            return self.build_tree_raxml(*args, **kwargs)
102        if algorithm == 'fasttree':
103            return self.build_tree_fast(*args, **kwargs)
104
105    def build_tree_raxml(self,
106                         new_path    = None,
107                         seq_type    = 'nucl' or 'prot',
108                         num_threads = None,
109                         free_cores  = 2,
110                         keep_dir    = False):
111        """Make a tree with RAxML."""
112        # Check output #
113        if new_path is None: new_path = self.prefix_path + '.tree'
114        # What model to choose #
115        if seq_type == 'nucl': model = "GTRGAMMA"
116        if seq_type == 'prot': model = "PROTGAMMAJTTF"
117        # Threads #
118        if num_threads is None:
119            num_threads = multiprocessing.cpu_count() - free_cores
120        else:
121            num_threads = int(num_threads) - free_cores
122        num_threads = max(1, num_threads)
123        # Run it #
124        temp_dir = new_temp_dir()
125        sh.raxml811('-m', model, "-T", num_threads, '-p', 1, '-s', self.path,
126                    '-n', 'tree', '-w', temp_dir, '-f', 'a', '-x', 1, '-N',
127                    'autoMR')
128        # Move into place #
129        if keep_dir:
130            shutil.rmtree(new_path)
131            shutil.move(temp_dir, new_path)
132        if not keep_dir:
133            shutil.move(temp_dir + 'RAxML_bestTree.tree', new_path)
134        # Return #
135        return FilePath(new_path)
136
137    def build_tree_fast(self, new_path=None, seq_type='nucl' or 'prot'):
138        """Make a tree with FastTree. Names will be truncated however."""
139        # Check output #
140        if new_path is None: new_path = self.prefix_path + '.tree'
141        # Command #
142        command_args = []
143        if seq_type == 'nucl': command_args += ['-nt']
144        command_args += ['-gamma']
145        command_args += ['-out', new_path]
146        command_args += [self.path]
147        # Run it #
148        sh.FastTree(*command_args)
149        # Return #
150        return FilePath(new_path)
