 37    def count_total(self):
 38        # Message for debugging purposes #
 39        if self.debug:
 40            print("-> counting reads in `%s`" % self.path)
 41        # Local helper #
 42        import subprocess
 43        def shell_output(cmd):
 44            output = subprocess.check_output(cmd, shell=True, text=True)
 45            return output.strip()
 46        # Case when we are not compressed #
 47        if not self.gzipped:
 48            count = int(shell_output("cat %s | wc -l" % self.path))
 49        # If we are gzipped, we can just use zcat or gzcat on macOS #
 50        else:
 51            program = 'gzcat' if sys.platform != 'linux' else 'zcat'
 52            command = "%s %s | wc -l" % (program, self.path)
 53            count = int(shell_output(command))
 54        # Check it is a multiple of four #
 55        assert count % 4 == 0
 56        # Return #
 57        return int(count / 4)

 61        """
 62        Returns a single float.
 63        Quite slow and inefficient computationally.
 64        Think of a better implementation?
 65        """
 66        mean = average(s for r in self for s in
 67                       r.letter_annotations["phred_quality"])
 68        return mean

106    #-------------------------------- PHRED Format ---------------------------#
107    def guess_phred_format(self):
108        """
109        Guess the PHRED score format. The gold standard is the first one, aka
110        the one called "Sanger". Sanger encoding is exactly equivalent to
111        Illumina-1.8 encoding.
112        In other words, they finally gave up with their alternative standards.
113        """
114        # Possibilities #
115        self.intervals = {
116            'Sanger':       (33,  74),    # <- This one is the gold standard
117            'Solexa':       (59, 105),    # <- This one is rare
118            'Illumina-1.3': (64, 105),    # <- These were abandoned after they wised up.
119            'Illumina-1.5': (67, 105),    # <- These were abandoned after they wised up.
120        }
121        # Initialize variables #
122        glob_min  = 9999
123        glob_max  = 0
124        take_next = False
125        # Stop after #
126        max_sequences_to_consider = 2000000
127        # Error message #
128        msg = "Multiple PHRED encodings possible for '%s'.\n"
129        # Loop #
130        self.open()
131        for i, line in enumerate(self.handle):
132            # Are we on the PHRED line? #
133            if not take_next:
134                take_next = line.startswith('+')
135                continue
136            # Now we are on the PHRED line! #
137            take_next = False
138            current_min, current_max = self.get_qual_range(line.rstrip('\n'))
139            if current_min < glob_min or current_max > glob_max:
140                glob_min = min(current_min, glob_min)
141                glob_max = max(current_max, glob_max)
142                valid_encodings = [e for e, r in self.intervals.items()
143                                   if glob_min >= r[0] and glob_max <= r[1]]
144                if len(valid_encodings) == 0:
145                    msg = "Illegal PHRED encoding for '%s'.\n"
146                    break
147                if len(valid_encodings) == 1:
148                    return valid_encodings[0]
149            # Stop condition #
150            if i >= max_sequences_to_consider: break
151        # It didn't work #
152        msg += "Maximum detected value is: %s.\n"
153        msg += "Minimum detected value is: %s.\n"
154        msg += "Possibilities are: %s."
155        msg  = msg % (self, glob_max, glob_min, ' or '.join(valid_encodings))
156        raise Exception(msg)
157
158    def get_qual_range(self, phred_string):
159        """
160        >>> self.get_qual_range("DLXYXXRXWYYTPMLUUQWTXTRSXSWMDMTRNDNSMJFJFFRMV")
161        (68, 89)
162        """
163        values = [ord(char) for char in phred_string]
164        return min(values), max(values)

167    def phred_13_to_18(self, new_path=None, in_place=True):
168        """Illumina-1.3 format conversion to Illumina-1.8 format via BioPython."""
169        # New file #
170        if new_path is None:
171            new_fastq = self.__class__(new_temp_path(suffix=self.extension))
172        else:
173            new_fastq = self.__class__(new_path)
174        # Do it #
175        self.format = 'fastq-illumina'
176        new_fastq.open('w')
177        new_fastq.handle.writelines(seq.format('fastq-sanger') for seq in self)
178        new_fastq.close()
179        self.format = 'fastq-sanger'
180        # Return #
181        if in_place:
182            os.remove(self.path)
183            shutil.move(new_fastq, self.path)
184            return self
185        else: return new_fastq

187    def phred_13_to_18_sed(self, new_path=None, in_place=True):
188        """Illumina-1.3 format conversion to Illumina-1.8 format via sed (faster)."""
189        # String #
190        sed_command = r"""4~4y/@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\\]^_`abcdefghi/!"#$%&'\''()*+,-.\/0123456789:;<=>?@ABCDEFGHIJ/"""
191        # Faster with bash utilities #
192        if in_place is True:
193            sh.sed('-i', sed_command, self.path)
194            return self
195        # New file #
196        if new_path is None: new_fastq = self.__class__(new_temp_path())
197        else:                new_fastq = self.__class__(new_path)
198        sh.sed(sed_command + " " + new_fastq, self.path)
199        return new_fastq
