17class SplitableFASTA(FASTA):
18    """
19    A FASTA file which you can split into chunks. Either you give the number
20    of parts you want to generate, or you can give a target size in bytes for
21    each part.
22    """
23
24    def __init__(self, path, num_parts=None, part_size=None, base_dir=None):
25        # Basic #
26        self.path = path
27        # Directory #
28        if base_dir is None: self.base_dir = path + '.parts/'
29        else:                self.base_dir = base_dir
30        # Num parts #
31        if num_parts is not None: self.num_parts = num_parts
32        # Special module #
33        import humanfriendly
34        # Evaluate size #
35        if part_size is not None:
36            self.bytes_target = humanfriendly.parse_size(part_size)
37            self.num_parts = int(math.ceil(self.count_bytes / self.bytes_target))
38        # Make parts #
39        self.make_name = lambda i: self.base_dir + "%03d/part.fasta" % i
40        self.parts = [FASTA(self.make_name(i))
41                      for i in range(1, self.num_parts+1)]
42        # Give a number to each part #
43        for i, part in enumerate(self.parts): part.num = i
44
45    @property
46    def status(self):
47        """Has the splitting been done already?"""
48        if all(os.path.exists(p.path) for p in self.parts): return True
49        return False
50
51    def run(self):
52        # Clean up #
53        for i in range(1, sys.maxsize):
54            dir_path = self.base_dir + "%03d/" % i
55            if os.path.exists(dir_path): shutil.rmtree(dir_path)
56            else: break
57        # Case only one part #
58        if len(self.parts) == 1:
59            self.parts[0].directory.create(safe=True)
60            self.link_to(self.parts[0])
61            return
62        # Compute number of sequences #
63        self.seqs_per_part = int(math.floor(self.count / self.num_parts))
64        # Prepare #
65        for part in self.parts: part.create()
66        # Do the job #
67        seqs = self.parse()
68        for part in self.parts:
69            for i in range(self.seqs_per_part):
70                part.add_seq(seqs.next())
71        # The final sequences go to the last part #
72        for seq in seqs: part.add_seq(seq)
73        # Clean up #
74        for part in self.parts: part.close()
